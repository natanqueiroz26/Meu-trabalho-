function abrirForm() { carregarSelectReparos(); // Apenas limpa a lista se não estiver em modo de edição if (edicaoIdx === -1) { itensTempAdicionados = []; atualizarListaTempVisual(); } let f = document.getElementById('form-adicionar'); document.getElementById('f-d').value = new Date().toISOString().split('T')[0]; document.getElementById('f-cli').value = ''; document.getElementById('f-kite').value = ''; document.getElementById('f-gas').value = 0; f.style.display = 'block'; } function editar(i) { let item = db.reparos[i]; edicaoIdx = i; // Abre o formulário let f = document.getElementById('form-adicionar'); carregarSelectReparos(); // Preenche os campos de texto com os dados salvos document.getElementById('f-d').value = item.data; document.getElementById('f-cli').value = item.cliente || ''; document.getElementById('f-kite').value = item.kite || ''; document.getElementById('f-bru').value = (item.bruto || 0).toFixed(2); document.getElementById('f-gas').value = (item.gasto || 0).toFixed(2); // Mantém os itens que já estavam na lista itensTempAdicionados = []; if (item.ser) { // Converte o texto salvo (ex: "1x Rasgo, 2x Furo") de volta em itens da lista let partes = item.ser.split(', '); partes.forEach(p => { let match = p.match(/^(\d+)x\s+(.+)$/); if (match) { let qtd = parseInt(match[1]); let nome = match[2]; let itemTb = db.tabelaPrecos.find(t => t.nome === nome); let preco = itemTb ? itemTb.preco : 0; itensTempAdicionados.push({ nome: nome, qtd: qtd, valorUnit: preco, subtotal: preco * qtd }); } }); } atualizarListaTempVisual(); f.style.display = 'block'; } function salvar() { let d = document.getElementById('f-d').value; if(!d) return alert('Selecione a data'); let cliente = document.getElementById('f-cli').value || 'Cliente não informado'; let kite = document.getElementById('f-kite').value || 'Kite não informado'; let bruto = parseFloat(document.getElementById('f-bru').value) || 0; let gasto = parseFloat(document.getElementById('f-gas').value) || 0; // Se houver itens na lista temporária, atualiza a descrição do serviço. // Se não alterou a lista, mantém a descrição que já existia. let resumoServicos = ''; if (itensTempAdicionados.length > 0) { resumoServicos = itensTempAdicionados.map(i => `${i.qtd}x ${i.nome}`).join(', '); } else if (edicaoIdx >= 0 && db.reparos[edicaoIdx].ser) { resumoServicos = db.reparos[edicaoIdx].ser; } else { resumoServicos = 'Reparo Avulso'; } let reg = { data: d, cliente: cliente, kite: kite, ser: resumoServicos, bruto: bruto, gasto: gasto, pago: false, valorMeu: (bruto - gasto) / 2 }; if(edicaoIdx >= 0) { reg.pago = db.reparos[edicaoIdx].pago || false; db.reparos[edicaoIdx] = reg; } else { db.reparos.push(reg); } db.reparos.sort((a,b) => new Date(b.data) - new Date(a.data)); localStorage.setItem(CHAVE, JSON.stringify(db)); fecharForm(); render(); }
